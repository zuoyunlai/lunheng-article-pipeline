    """返回 (passed, hit_list)；hit_list 为命中的污染源字符串"""
    import re
    with open(final_md_path, encoding='utf-8') as f:
        head = '\n'.join(f.readlines()[:10])
    forbidden = [
        r'v\d+\.\d+',                          # 版本号

        r'T6 批判报告|T7 审计|G14 检测|修订说明 v\d',

```python
def check_citation_ordering(final_md_path: str) -> tuple[bool, dict]:
    """返回 (passed, report)；
    passed = 正文引用顺序 与 文末列表顺序 完全闭环（双向无错位）"""
    import re
    with open(final_md_path, encoding='utf-8') as f:
        text = f.read()

    # 1. 抽取正文所有引用编号 [Cxx]/[Dxx]/[Lxx]/[先xx]/[C-主xx]
    body_refs = re.findall(r'\[(C-\u4e3b?\d+|D\d+|L\d+|\u5148\d+)\]', text)

    # 2. 定位「文末附录」起点（## 引用来源 / ## 参考文献 / ## 先行者文献）
    appendix_start = -1
    for marker in ['## 引用来源', '## 参考文献', '## 先行者文献']:
        idx = text.find(marker)
        if idx > appendix_start:
            appendix_start = idx

    if appendix_start < 0:
        return (False, {'error': 'no_appendix_found'})

    appendix = text[appendix_start:]

    # 3. 抽取文末列表中每个编号（按出现顺序）
    appendix_refs = re.findall(r'\[(C-\u4e3b?\d+|D\d+|L\d+|\u5148\d+)\]', appendix)

    # 4. 闭环验证：body_refs 去重后顺序 ∩ appendix_refs 顺序 应一致
    seen = set()
    body_unique_ordered = []
    for r in body_refs:
        if r not in seen:
            seen.add(r)
            body_unique_ordered.append(r)

    # 缺失：正文引用了但文末没有
    missing_in_appendix = [r for r in body_unique_ordered if r not in appendix_refs]
    # 悬空：文末有但正文没引用
    unused_in_body = [r for r in appendix_refs if r not in body_unique_ordered]

    passed = (len(missing_in_appendix) == 0 and len(unused_in_body) == 0)

    return (passed, {
        'body_unique_ordered': body_unique_ordered,
        'appendix_ordered': appendix_refs,
        'missing_in_appendix': missing_in_appendix,
        'unused_in_body': unused_in_body,
    })
```

def check_charts(final_md_path: str, project_dir: str) -> tuple[bool, dict]:
    """返回 (passed, report)"""
    import re, os, glob
    with open(final_md_path, encoding='utf-8') as f:
        text = f.read()

    # 5.1 图件文件数
    svg_files = glob.glob(os.path.join(project_dir, 'final', '图件', '*.svg'))
    svg_count = len(svg_files)

    # 5.2 正文图引用数
    figure_slots = len(re.findall(r'\[图\d+[：:]', text))
    md_image_refs = len(re.findall(r'!\[[^\]]*\]\([^)]*\)', text))
    body_ref_count = figure_slots + md_image_refs

    # 5.3 图位置内嵌 = 图引用在文中分布的标准差（粗略启发）
    # 简化：仅检查文末是否有 ≥3 张图堆放
    lines = text.splitlines()
    last_5th = len(lines) * 4 // 5
    tail_refs = sum(1 for line in lines[last_5th:]
                    if re.search(r'\[图\d+[：:]', line) or '![' in line)
    inlined = tail_refs < 3

    # 5.4 数据溯源（每张图件文件含 [Dxx] 或 [Lxx]）
    data_sources = []
    for svg in svg_files:
        with open(svg, encoding='utf-8') as f:
            svg_text = f.read()
        data_sources.append(bool(re.search(r'\[(D\d+|L\d+)\]', svg_text)))

    # 5.5 图序号闭环
    body_fig_nums = sorted(set(int(m.group(1))
        for m in re.finditer(r'\[图 (\d+)\]', text)))
    svg_fig_nums = sorted(set(int(re.search(r'(\d+)', os.path.basename(f)).group(1))
        for f in svg_files if re.search(r'(\d+)', os.path.basename(f))))
    seq_match = body_fig_nums == svg_fig_nums

    passed = (mmd_count >= 5 and body_ref_count >= 5 and inlined
              and all(data_sources) and seq_match)
    return (passed, {
        'mmd_count': mmd_count,
        'body_ref_count': body_ref_count,
        'inlined': inlined,
        'data_sources': data_sources,
        'seq_match': seq_match,
    })
